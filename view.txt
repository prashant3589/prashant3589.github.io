# ============================================
# Windows WiFi Diagnostic & Repair Script
# Author: ChatGPT
# Version: 1.0
# ============================================

Write-Host ""
Write-Host "==========================================" -ForegroundColor Cyan
Write-Host " Windows WiFi Diagnostic & Repair Tool"
Write-Host "==========================================" -ForegroundColor Cyan
Write-Host ""

# Check Administrator
$currentUser = [Security.Principal.WindowsIdentity]::GetCurrent()
$principal = New-Object Security.Principal.WindowsPrincipal($currentUser)

if (-not $principal.IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator))
{
    Write-Host "Please run PowerShell as Administrator." -ForegroundColor Red
    exit
}

Write-Host "Running with Administrator privileges." -ForegroundColor Green

# ---------------------------------------------
# Network Adapters
# ---------------------------------------------
Write-Host ""
Write-Host "Network Adapters"
Write-Host "----------------"

Get-NetAdapter | Format-Table Name, InterfaceDescription, Status, MacAddress

# ---------------------------------------------
# Find WiFi Adapter
# ---------------------------------------------

$wifi = Get-NetAdapter |
Where-Object {
    $_.InterfaceDescription -match "Wireless|Wi-Fi|802.11|Intel|Realtek|Qualcomm|MediaTek"
}

if($wifi)
{
    Write-Host ""
    Write-Host "WiFi Adapter Found:" -ForegroundColor Green

    $wifi | Format-Table Name, Status, InterfaceDescription

    foreach($adapter in $wifi)
    {
        if($adapter.Status -eq "Disabled")
        {
            Write-Host ""
            Write-Host "Enabling Adapter..." -ForegroundColor Yellow

            Enable-NetAdapter `
                -Name $adapter.Name `
                -Confirm:$false
        }
    }
}
else
{
    Write-Host ""
    Write-Host "No WiFi adapter detected!" -ForegroundColor Red
    Write-Host "Possible Causes:"
    Write-Host " - Driver Missing"
    Write-Host " - Hardware Disabled"
    Write-Host " - BIOS Wireless Disabled"
}

# ---------------------------------------------
# WLAN Service
# ---------------------------------------------

Write-Host ""
Write-Host "Checking WLAN AutoConfig"

$service = Get-Service WlanSvc -ErrorAction SilentlyContinue

if($service)
{
    if($service.Status -ne "Running")
    {
        Write-Host "Starting WLAN Service..." -ForegroundColor Yellow

        Set-Service WlanSvc -StartupType Automatic

        Start-Service WlanSvc
    }

    Write-Host "WLAN Service Running" -ForegroundColor Green
}
else
{
    Write-Host "WLAN Service Not Found!" -ForegroundColor Red
}

# ---------------------------------------------
# Wireless Interface
# ---------------------------------------------

Write-Host ""
Write-Host "Wireless Interface"
Write-Host "------------------"

netsh wlan show interfaces

Write-Host ""
Write-Host "Wireless Driver"
Write-Host "---------------"

netsh wlan show drivers

# ---------------------------------------------
# IP Configuration
# ---------------------------------------------

Write-Host ""
Write-Host "IP Configuration"
Write-Host "----------------"

ipconfig /all

# ---------------------------------------------
# Network Repair
# ---------------------------------------------

Write-Host ""
Write-Host "Resetting Network Stack..."

netsh winsock reset

netsh int ip reset

ipconfig /flushdns

# ---------------------------------------------
# SFC
# ---------------------------------------------

Write-Host ""
Write-Host "Running System File Checker"

sfc /scannow

# ---------------------------------------------
# Restart Adapter
# ---------------------------------------------

if($wifi)
{
    foreach($adapter in $wifi)
    {
        Write-Host ""
        Write-Host "Restarting Adapter $($adapter.Name)"

        Disable-NetAdapter `
            -Name $adapter.Name `
            -Confirm:$false

        Start-Sleep 3

        Enable-NetAdapter `
            -Name $adapter.Name `
            -Confirm:$false
    }
}

Write-Host ""
Write-Host "==========================================" -ForegroundColor Cyan
Write-Host "Diagnostics Completed"
Write-Host "==========================================" -ForegroundColor Cyan

Write-Host ""
Write-Host "If WiFi is still missing:"
Write-Host "1. Check Device Manager."
Write-Host "2. Scan for Hardware Changes."
Write-Host "3. Install OEM WiFi Driver."
Write-Host "4. Check BIOS Wireless Enable."
Write-Host "5. Reboot Computer."

Write-Host ""
